cmake_minimum_required(VERSION 3.24)
project(context_read_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(FAISS_SOURCE /home/dev_path/ceph/faiss CACHE PATH "FAISS source tree")
set(FAISS_ENABLE_GPU OFF CACHE BOOL "" FORCE)
set(FAISS_ENABLE_PYTHON OFF CACHE BOOL "" FORCE)
set(FAISS_ENABLE_EXTRAS OFF CACHE BOOL "" FORCE)
set(FAISS_ENABLE_MKL OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(FAISS_OPT_LEVEL avx2 CACHE STRING "" FORCE)
add_subdirectory(${FAISS_SOURCE} faiss-build EXCLUDE_FROM_ALL)
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)
add_executable(context_bench src/main.cpp)
target_compile_options(context_bench PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(context_bench PRIVATE faiss_avx2 OpenMP::OpenMP_CXX Threads::Threads)
